# OpenBot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build OpenBot v1: a self-hosted web app hosting multiple persistent AI bots that talk to humans and each other in shared threads, use pluggable tools, keep LangMem memory, pause for human approval, and are reachable over REST/SSE/webhooks.

**Architecture:** One FastAPI process holds the REST API, an SSE event bus, and an in-process asyncio run queue. Each bot run builds a fresh LangChain `create_agent` with the bot's model, tools, LangMem memory tools, and a LangGraph checkpointer for interrupts; thread history from SQL is rendered from the bot's point of view. A React/Vite frontend talks to the API and streams events.

**Tech Stack:** Python 3.12+, uv, FastAPI, SQLAlchemy 2 (async, aiosqlite), Alembic, LangChain 1.4 (`create_agent`, `ToolRuntime`, `HumanInTheLoopMiddleware`), LangGraph 1.2 (`interrupt`, `AsyncSqliteSaver`, `AsyncSqliteStore`), LangMem 0.0.30, langchain-openai, langchain-anthropic, httpx, pytest/pytest-asyncio, React 19, Vite, TypeScript, Tailwind v4, TanStack Query, react-router.

**Spec:** `docs/superpowers/specs/2026-09-13-openbot-design.md`

## Global Constraints

- Python `>=3.12`; backend lives in `backend/`, package `openbot`, managed with `uv`. Run backend commands from `backend/` with `uv run ...`.
- Frontend lives in `frontend/`, managed with `pnpm`. Node 24.
- Standard SQL only through SQLAlchemy `JSON`, `String`, `Text`, `Boolean`, `Integer`, `DateTime(timezone=True)`. No SQLite-specific SQL in application code. `DATABASE_URL` default `sqlite+aiosqlite:///./openbot.db`.
- Providers: `openai`, `anthropic`, `openrouter`, `xai`. Keys from env: `OPENAI_API_KEY`, `ANTHROPIC_API_KEY`, `OPENROUTER_API_KEY`, `XAI_API_KEY`.
- Config env names and defaults exactly as spec section 12 plus `EMBEDDING_DIMS=1536`, `CORS_ORIGINS`, `FRONTEND_DIST`, `WEBHOOK_RETRY_DELAYS=5,30,120`.
- Bot handle regex `^[a-z0-9_-]{2,32}$`. Mention syntax `@handle`.
- Run statuses: `queued|running|waiting_human|completed|failed|cancelled`. Run event types: `text|tool_call|tool_result|interrupt|resumed|error|message`.
- Message sender kinds: `human|bot|external|system`. System messages never trigger runs. A bot never triggers itself.
- Defaults: `MAX_CONCURRENT_RUNS=4`, `MAX_BOT_HOPS=20`, `HISTORY_TOKEN_BUDGET=24000`, `HISTORY_MAX_MESSAGES=80`, `MEMORY_REFLECTION_DELAY=30`.
- LangSmith via env passthrough (`LANGSMITH_TRACING`, `LANGSMITH_API_KEY`, `LANGSMITH_PROJECT`, `LANGSMITH_ENDPOINT`); `.env` is loaded into `os.environ` with `python-dotenv` at startup so the langsmith SDK sees it.
- Every task ends with green tests and a commit. Commit messages end with the two attribution lines given in the session reminder.
- Tests never call real providers except the opt-in smoke tests in Task 24.

```python
# backend/openbot/main.py
from __future__ import annotations

from contextlib import asynccontextmanager

from dotenv import load_dotenv
from fastapi import APIRouter, FastAPI

from openbot.config import Settings, get_settings


def create_app(settings: Settings | None = None, services=None) -> FastAPI:
    load_dotenv()
    settings = settings or get_settings()

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        yield

    app = FastAPI(title="OpenBot", lifespan=lifespan)
    app.state.settings = settings
    app.state.services = services

    api = APIRouter(prefix="/api/v1")

    @api.get("/health")
    async def health():
        return {"status": "ok"}

    app.include_router(api)
    return app


app = create_app()
```

```python
OPEN_RUN_STATUSES = ("queued", "running", "waiting_human")


def new_id() -> str:
    return str(uuid.uuid4())


def utcnow() -> datetime:
    return datetime.now(timezone.utc)


class Base(DeclarativeBase):
    pass


class TimestampMixin:
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=utcnow, nullable=False)
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), default=utcnow, onupdate=utcnow, nullable=False

class Bot(TimestampMixin, Base):
```

```python
# backend/openbot/db/session.py
from __future__ import annotations

import asyncio
from pathlib import Path

from alembic import command
from alembic.config import Config
from sqlalchemy import event
from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession, async_sessionmaker, create_async_engine
from sqlalchemy.pool import StaticPool

from openbot.db.models import Base


def make_engine(url: str) -> AsyncEngine:
    kwargs: dict = {}
    if url.startswith("sqlite"):
        kwargs["connect_args"] = {"check_same_thread": False}
        if ":memory:" in url:
            kwargs["poolclass"] = StaticPool
    engine = create_async_engine(url, **kwargs)
    if url.startswith("sqlite"):
        @event.listens_for(engine.sync_engine, "connect")
        def _pragmas(dbapi_conn, _):
            cur = dbapi_conn.cursor()
            cur.execute("PRAGMA foreign_keys=ON")
            cur.execute("PRAGMA journal_mode=WAL")
            cur.execute("PRAGMA busy_timeout=5000")
            cur.close()
    return engine


def make_session_factory(engine: AsyncEngine) -> async_sessionmaker[AsyncSession]:
    return async_sessionmaker(engine, expire_on_commit=False)


async def create_all(engine: AsyncEngine) -> None:
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)


def _alembic_config(database_url: str) -> Config:
    here = Path(__file__).resolve().parent
    cfg = Config()
    cfg.set_main_option("script_location", str(here / "migrations"))
    cfg.set_main_option("sqlalchemy.url", database_url)
    return cfg


async def run_migrations(database_url: str) -> None:
    cfg = _alembic_config(database_url)
    await asyncio.to_thread(command.upgrade, cfg, "head")
```

```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from openbot.api.deps import get_services, get_session
from openbot.api.schemas import BotCreate, BotOut, BotUpdate
from openbot.db.models import OPEN_RUN_STATUSES, Bot, Run
from openbot.services import Services

router = APIRouter(prefix="/bots", tags=["bots"])


def _validate_tools(services: Services, tool_names: list[str], approval_tools: list[str]) -> None:
    if services.registry is None:
        return
    unknown = [t for t in tool_names if not services.registry.has(t)]
    if unknown:
        raise HTTPException(422, f"unknown tools: {unknown}")
    extra = [t for t in approval_tools if t not in tool_names]
    if extra:
        raise HTTPException(422, f"approval_tools must be a subset of tool_names: {extra}")


async def _get_or_404(session: AsyncSession, bot_id: str) -> Bot:
    bot = await session.get(Bot, bot_id)
    if not bot:
        raise HTTPException(404, "bot not found")
    return bot


@router.get("", response_model=list[BotOut])
async def list_bots(session: AsyncSession = Depends(get_session)):
    return (await session.execute(select(Bot).order_by(Bot.created_at))).scalars().all()


@router.post("", response_model=BotOut, status_code=201)
async def create_bot(body: BotCreate, session: AsyncSession = Depends(get_session),
                     services: Services = Depends(get_services)):
    _validate_tools(services, body.tool_names, body.approval_tools)
    if (await session.execute(select(Bot).where(Bot.handle == body.handle))).scalar_one_or_none():
        raise HTTPException(409, "handle already exists")
    bot = Bot(**body.model_dump())
    session.add(bot)
    await session.commit()
    return bot


@router.get("/{bot_id}", response_model=BotOut)
async def get_bot(bot_id: str, session: AsyncSession = Depends(get_session)):
    return await _get_or_404(session, bot_id)


@router.patch("/{bot_id}", response_model=BotOut)
async def update_bot(bot_id: str, body: BotUpdate, session: AsyncSession = Depends(get_session),
                     services: Services = Depends(get_services)):
    bot = await _get_or_404(session, bot_id)
    data = body.model_dump(exclude_unset=True)
    tool_names = data.get("tool_names", bot.tool_names)
    approval = data.get("approval_tools", bot.approval_tools)
    _validate_tools(services, tool_names, approval)
    if "handle" in data and data["handle"] != bot.handle:
        if (await session.execute(select(Bot).where(Bot.handle == data["handle"]))).scalar_one_or_none():
            raise HTTPException(409, "handle already exists")
    for k, v in data.items():
        setattr(bot, k, v)
    await session.commit()
    return bot


@router.delete("/{bot_id}", status_code=204)
async def delete_bot(bot_id: str, session: AsyncSession = Depends(get_session)):
    bot = await _get_or_404(session, bot_id)
    active = (await session.execute(
        select(Run).where(Run.bot_id == bot_id, Run.status.in_(OPEN_RUN_STATUSES)))).first()
    if active:
        raise HTTPException(409, "bot has active runs")
    await session.delete(bot)
    await session.commit()
```

```python
        if owned:
            app.state.services = await build_services(settings)
        yield

    app.include_router(public)
    app.include_router(api)
    return app
```

```python
# backend/openbot/tools/registry.py
from __future__ import annotations

import importlib.util
import logging
import sys
from dataclasses import dataclass
from pathlib import Path

from langchain_core.tools import BaseTool

log = logging.getLogger(__name__)


@dataclass
class ToolSpec:
    name: str
    description: str
    source: str
    tool: BaseTool


class ToolRegistry:
    def __init__(self) -> None:
        self._tools: dict[str, ToolSpec] = {}
        self.load_errors: list[dict] = []

    def register(self, tool: BaseTool, source: str = "builtin") -> None:
        if tool.name in self._tools:
            log.warning("tool %s from %s overrides %s", tool.name, source, self._tools[tool.name].source)
        self._tools[tool.name] = ToolSpec(tool.name, tool.description or "", source, tool)

    def has(self, name: str) -> bool:
        return name in self._tools

    def get(self, name: str) -> BaseTool:
        return self._tools[name].tool

    def resolve(self, names: list[str]) -> list[BaseTool]:
        unknown = [n for n in names if n not in self._tools]
        if unknown:
            raise KeyError(f"unknown tools: {unknown}")
        return [self._tools[n].tool for n in names]

    def specs(self) -> list[ToolSpec]:
        return list(self._tools.values())

    def load_plugins(self, tools_dir: Path) -> None:
        if not tools_dir.is_dir():
            return
        for path in sorted(tools_dir.glob("*.py")):
            if path.name.startswith("_"):
                continue
            mod_name = f"openbot_plugins.{path.stem}"
            try:
                spec = importlib.util.spec_from_file_location(mod_name, path)
                module = importlib.util.module_from_spec(spec)  # type: ignore[arg-type]
                sys.modules[mod_name] = module
                spec.loader.exec_module(module)  # type: ignore[union-attr]
            except Exception as e:  # noqa: BLE001
                log.exception("failed to load plugin %s", path)
                self.load_errors.append({"file": str(path), "error": f"{type(e).__name__}: {e}"})
                continue
            found = 0
            for obj in vars(module).values():
                if isinstance(obj, BaseTool):
                    self.register(obj, source=str(path))
                    found += 1
            log.info("loaded %d tools from %s", found, path)


def build_registry(settings) -> ToolRegistry:
    from openbot.tools.builtin import SELECTABLE_TOOLS

    reg = ToolRegistry()
    for t in SELECTABLE_TOOLS:
        reg.register(t)
    reg.load_plugins(Path(settings.tools_dir))
    return reg
```

```python
OUTPUT_CAP = 20000


def resolve_in_workspace(root: Path, path: str | None) -> Path:
    root = root.resolve()

    if target != root and root not in target.parents:
```

```python
import asyncio

from langchain.tools import ToolRuntime, tool

from openbot.tools.builtin.workspace import cap, resolve_in_workspace
from openbot.tools.context import RunContext


@tool
async def run_shell(command: str, runtime: ToolRuntime[RunContext], cwd: str | None = None,
                    timeout: int = 120) -> str:
    """Run a shell command (bash) inside the workspace. Use it for git, gh, tests, builds.
    `cwd` is relative to the workspace root. Returns stdout, stderr and the exit code."""
    try:
        workdir = resolve_in_workspace(runtime.context.workspace_root, cwd)
    except ValueError as e:
        return f"error: {e}"
    workdir.mkdir(parents=True, exist_ok=True)
    proc = await asyncio.create_subprocess_exec(
        "bash", "-lc", command, cwd=str(workdir),
        stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE)
    try:
        out, err = await asyncio.wait_for(proc.communicate(), timeout=timeout)
    except asyncio.TimeoutError:
        proc.kill()
        await proc.wait()
        return f"error: command timed out after {timeout}s"
    parts = [f"exit code: {proc.returncode}"]
    if out:
        parts.append("stdout:\n" + out.decode(errors="replace"))
    if err:
        parts.append("stderr:\n" + err.decode(errors="replace"))
    return cap("\n".join(parts))
```

```python
import os

from langchain.tools import ToolRuntime, tool

from openbot.tools.builtin.workspace import cap, resolve_in_workspace
from openbot.tools.context import RunContext


@tool
async def read_file(path: str, runtime: ToolRuntime[RunContext]) -> str:
    """Read a UTF-8 text file at a path relative to the workspace root."""
    try:
        p = resolve_in_workspace(runtime.context.workspace_root, path)
        return cap(p.read_text(encoding="utf-8", errors="replace"))
    except (ValueError, OSError) as e:
        return f"error: {e}"


@tool
async def write_file(path: str, content: str, runtime: ToolRuntime[RunContext]) -> str:
    """Write (create or overwrite) a UTF-8 text file at a path relative to the workspace root."""
    try:
        p = resolve_in_workspace(runtime.context.workspace_root, path)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(content, encoding="utf-8")
        return f"wrote {len(content)} chars to {path}"
    except (ValueError, OSError) as e:
        return f"error: {e}"


@tool
async def list_files(runtime: ToolRuntime[RunContext], path: str = ".", depth: int = 2) -> str:
    """List files under a directory (relative to the workspace root) up to `depth` levels.
    Skips .git, node_modules, .venv and __pycache__."""
    try:
        root = resolve_in_workspace(runtime.context.workspace_root, path)
    except ValueError as e:
        return f"error: {e}"
    if not root.exists():
        return f"error: {path} does not exist"
    skip = {".git", "node_modules", ".venv", "__pycache__", "dist"}
    lines: list[str] = []
    base_depth = len(root.parts)
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = sorted(d for d in dirnames if d not in skip)
        level = len(os.path.normpath(dirpath).split(os.sep)) - base_depth
        if level >= depth:
            dirnames[:] = []
        rel = os.path.relpath(dirpath, root)
        for f in sorted(filenames):
            lines.append(f if rel == "." else f"{rel}/{f}")
    return cap("\n".join(lines) or "(empty)")
```

```python
import httpx
from bs4 import BeautifulSoup
from langchain.tools import tool

from openbot.tools.builtin.workspace import cap


@tool
async def http_request(method: str, url: str, headers: dict[str, str] | None = None,
                       body: str | None = None) -> str:
    """Make an HTTP request. Returns status, response headers and body (text, capped)."""
    try:
        async with httpx.AsyncClient(timeout=60, follow_redirects=True) as c:
            r = await c.request(method.upper(), url, headers=headers, content=body)
    except httpx.HTTPError as e:
        return f"error: {e}"
    hdrs = "\n".join(f"{k}: {v}" for k, v in r.headers.items())
    return cap(f"status: {r.status_code}\nheaders:\n{hdrs}\n\nbody:\n{r.text}")


@tool
async def fetch_url(url: str) -> str:
    """Fetch a web page and return its readable text content."""
    try:
        async with httpx.AsyncClient(timeout=60, follow_redirects=True) as c:
            r = await c.get(url)
    except httpx.HTTPError as e:
        return f"error: {e}"
    ctype = r.headers.get("content-type", "")
    if "html" not in ctype:
        return cap(r.text)
    soup = BeautifulSoup(r.text, "html.parser")
    for t in soup(["script", "style", "noscript"]):
        t.decompose()
    text = "\n".join(line.strip() for line in soup.get_text("\n").splitlines() if line.strip())
    return cap(text)
```

```python
# backend/openbot/runtime/router.py
from __future__ import annotations

import re

from openbot.db.models import Bot

MENTION_RE = re.compile(r"(?<![\w@.])@([a-z0-9_-]{2,32})(?![\w.-])")


def parse_mentions(content: str) -> list[str]:
    seen: list[str] = []
    for m in MENTION_RE.finditer(content):
        h = m.group(1)
        if h not in seen:
            seen.append(h)
    return seen


def resolve_targets(*, sender_kind: str, sender_bot_id: str | None, mentioned_handles: list[str],
                    to_handles: list[str], bots_by_handle: dict[str, Bot],
                    thread_bot_ids: list[str]) -> list[Bot]:
    if sender_kind == "system":
        return []
    ordered: list[Bot] = []
    for h in [*to_handles, *mentioned_handles]:
        bot = bots_by_handle.get(h)
        if bot and bot not in ordered:
            ordered.append(bot)
    if not ordered and len(thread_bot_ids) == 1:
        for bot in bots_by_handle.values():
            if bot.id == thread_bot_ids[0]:
                ordered.append(bot)
    return [b for b in ordered if b.enabled and b.id != sender_bot_id]
```

```python
# backend/openbot/runtime/bus.py
from __future__ import annotations

import asyncio
from contextlib import asynccontextmanager
from typing import AsyncIterator


class EventBus:
    def __init__(self, maxsize: int = 1000) -> None:
        self._subs: dict[asyncio.Queue, str | None] = {}
        self._maxsize = maxsize

    async def publish(self, event: str, thread_id: str | None, data: dict) -> None:
        item = {"event": event, "thread_id": thread_id, "data": data}
        for q, filt in list(self._subs.items()):
            if filt is not None and filt != thread_id:
                continue
            if q.full():
                try:
                    q.get_nowait()
                except asyncio.QueueEmpty:
                    pass
                self._put(q, {"event": "overflow", "thread_id": thread_id, "data": {"dropped": 1}})
            self._put(q, item)

    @staticmethod
    def _put(q: asyncio.Queue, item: dict) -> None:
        try:
            q.put_nowait(item)
        except asyncio.QueueFull:
            pass

    @asynccontextmanager
    async def subscribe(self, thread_id: str | None) -> AsyncIterator[asyncio.Queue]:
        q: asyncio.Queue = asyncio.Queue(maxsize=self._maxsize)
        self._subs[q] = thread_id
        try:
            yield q
        finally:
            self._subs.pop(q, None)
```

```python
    for p in PROVIDER_ORDER:
```

```python
from __future__ import annotations

from contextlib import asynccontextmanager
from pathlib import Path
from typing import AsyncIterator

import aiosqlite
from langgraph.checkpoint.base import BaseCheckpointSaver
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.store.base import BaseStore
from langgraph.store.memory import InMemoryStore

from openbot.config import Settings


def sqlite_path(url: str) -> str:
    return url.split("///", 1)[1]


def _index(settings: Settings, embeddings):
    if embeddings is None:
        return None
    return {"dims": settings.embedding_dims, "embed": embeddings, "fields": ["content"]}


@asynccontextmanager
async def open_langgraph_backends(settings: Settings, embeddings) -> AsyncIterator[tuple[BaseCheckpointSaver, BaseStore]]:
    url = settings.database_url
    index = _index(settings, embeddings)
    if url.startswith("sqlite"):
        path = sqlite_path(url)
        if path == ":memory:":
            yield InMemorySaver(), InMemoryStore(index=index)
            return
        from langgraph.checkpoint.sqlite.aio import AsyncSqliteSaver
        from langgraph.store.sqlite import AsyncSqliteStore
        lg_path = Path(path).with_suffix(".langgraph.db")
        lg_path.parent.mkdir(parents=True, exist_ok=True)
        async with aiosqlite.connect(lg_path) as c1, aiosqlite.connect(lg_path) as c2:
            for c in (c1, c2):
                await c.execute("PRAGMA journal_mode=WAL")
                await c.execute("PRAGMA busy_timeout=5000")
            saver = AsyncSqliteSaver(c1)
            await saver.setup()
            store = AsyncSqliteStore(c2, index=index)
            await store.setup()
            yield saver, store
        return
    if url.startswith("postgresql"):
        from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver
        from langgraph.store.postgres.aio import AsyncPostgresStore
        dsn = "postgresql://" + url.split("://", 1)[1]
        async with AsyncPostgresSaver.from_conn_string(dsn) as saver, \
                AsyncPostgresStore.from_conn_string(dsn, index=index) as store:
            await saver.setup()
            await store.setup()
            yield saver, store
        return
    raise ValueError(f"unsupported DATABASE_URL: {url}")
```

```python
# backend/openbot/runtime/memory.py
from __future__ import annotations

import asyncio
import logging

from langchain_core.messages import BaseMessage
from langchain_core.tools import BaseTool
from langgraph.store.base import BaseStore
from langmem import create_manage_memory_tool, create_memory_store_manager, create_search_memory_tool

from openbot.db.models import Bot, Message

log = logging.getLogger(__name__)


def bot_namespace(bot_id: str) -> tuple[str, ...]:
    return ("bots", bot_id, "memories")


def thread_namespace(thread_id: str) -> tuple[str, ...]:
    return ("threads", thread_id, "messages")


def memory_tools(bot_id: str, store: BaseStore) -> list[BaseTool]:
    ns = bot_namespace(bot_id)
    return [
        create_manage_memory_tool(ns, store=store,
            instructions="Save durable facts, preferences, decisions and lessons you will need in "
                         "future conversations. Update or delete memories that became wrong."),
        create_search_memory_tool(ns, store=store),
    ]


async def relevant_memories(store: BaseStore, bot_id: str, query: str, limit: int = 8) -> list[str]:
    items = await store.asearch(bot_namespace(bot_id), query=query or None, limit=limit)
    out = []
    for it in items:
        v = it.value
        out.append(str(v.get("content", v)) if isinstance(v, dict) else str(v))
    return out


async def index_message(store: BaseStore, message: Message) -> None:
    await store.aput(thread_namespace(message.thread_id), message.id, {
        "content": message.content, "sender": message.sender_name,
        "created_at": message.created_at.isoformat() if message.created_at else None,
        "message_id": message.id,
    })


async def recall(store: BaseStore, thread_id: str, query: str, limit: int = 10) -> list[dict]:
    items = await store.asearch(thread_namespace(thread_id), query=query, limit=limit)
    return [dict(it.value) for it in items]


class MemoryReflector:
    """Debounced background extraction of long-term memories after runs."""

    def __init__(self, services, delay: float) -> None:
        self.services = services
        self.delay = delay
        self._pending: dict[str, tuple[Bot, list[BaseMessage]]] = {}
        self._timers: dict[str, asyncio.Task] = {}

    def make_manager(self, bot: Bot):
        model = self.services.model_factory(bot)
        return create_memory_store_manager(model, namespace=bot_namespace(bot.id),
                                           store=self.services.store,
                                           enable_inserts=True, enable_deletes=False)

    def schedule(self, bot: Bot, messages: list[BaseMessage]) -> None:
        _, existing = self._pending.get(bot.id, (bot, []))
        self._pending[bot.id] = (bot, [*existing, *messages])
        if t := self._timers.pop(bot.id, None):
            t.cancel()
        self._timers[bot.id] = asyncio.create_task(self._later(bot.id))

    async def _later(self, bot_id: str) -> None:
        await asyncio.sleep(self.delay)
        await self._reflect(bot_id)

    async def _reflect(self, bot_id: str) -> None:
        self._timers.pop(bot_id, None)
        item = self._pending.pop(bot_id, None)
        if not item:
            return
        bot, messages = item
        try:
            await self.make_manager(bot).ainvoke({"messages": messages})
        except Exception:  # noqa: BLE001
            log.exception("memory reflection failed for bot %s", bot.handle)

    async def flush(self) -> None:
        for bot_id in list(self._pending):
            if t := self._timers.pop(bot_id, None):
                t.cancel()
            await self._reflect(bot_id)

    async def shutdown(self) -> None:
        for t in self._timers.values():
            t.cancel()
        self._timers.clear()
        self._pending.clear()
```

```python
from dataclasses import dataclass

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

@dataclass
class PostResult:
    message: Message
    runs: list[Run]
    unaddressed: bool


async def _bots_by_handle(session: AsyncSession) -> dict[str, Bot]:
    bots = (await session.execute(select(Bot))).scalars().all()
    return {b.handle: b for b in bots}


async def _thread_bot_ids(session: AsyncSession, thread_id: str) -> list[str]:
    rows = (await session.execute(select(ThreadParticipant.bot_id).where(
        ThreadParticipant.thread_id == thread_id, ThreadParticipant.kind == "bot"))).scalars().all()
    return [r for r in rows if r]


async def create_thread(services, session: AsyncSession, *, title: str, handles: list[str],
                        created_by_kind: str, created_by_bot_id: str | None = None,
                        external_ref: str | None = None) -> Thread:
    by_handle = await _bots_by_handle(session)
    unknown = [h for h in handles if h not in by_handle]
    if unknown:
        raise ValueError(f"unknown bot handles: {unknown}")
    thread = Thread(title=title, created_by_kind=created_by_kind, created_by_bot_id=created_by_bot_id,
                    external_ref=external_ref)
    session.add(thread)
    await session.flush()
    session.add(ThreadParticipant(thread_id=thread.id, kind="human", bot_id=None))
    bot_ids = {by_handle[h].id for h in handles}
    if created_by_bot_id:
        bot_ids.add(created_by_bot_id)
    for bid in bot_ids:
        session.add(ThreadParticipant(thread_id=thread.id, kind="bot", bot_id=bid))
    await session.commit()
    return thread


async def post_message(services, session: AsyncSession, *, thread_id: str, content: str, sender_kind: str,
                       sender_name: str, sender_bot_id: str | None = None, to_handles=(), hop: int = 0,
                       run_id: str | None = None, meta: dict | None = None) -> PostResult:
    thread = await session.get(Thread, thread_id)
    if thread is None:
        raise LookupError("thread not found")
    by_handle = await _bots_by_handle(session)
    unknown = [h for h in to_handles if h not in by_handle]
    if unknown:
        raise ValueError(f"unknown bot handles: {unknown}")
    mentioned = parse_mentions(content)
    thread_bot_ids = await _thread_bot_ids(session, thread_id)
    targets = resolve_targets(sender_kind=sender_kind, sender_bot_id=sender_bot_id, mentioned_handles=mentioned,
                              to_handles=list(to_handles), bots_by_handle=by_handle, thread_bot_ids=thread_bot_ids)
    explicit = bool(mentioned or to_handles)
    unaddressed = sender_kind != "system" and not targets and not explicit
    now = utcnow()
    msg = Message(thread_id=thread_id, sender_kind=sender_kind, sender_bot_id=sender_bot_id,
                  sender_name=sender_name, content=content,
                  mentions=[by_handle[h].id for h in mentioned if h in by_handle],
                  hop=hop, run_id=run_id, meta=meta or {}, created_at=now)
    session.add(msg)
    thread.last_message_at = now
    thread.updated_at = now
    if hop == 0 and sender_kind != "system":
        thread.hop_limit_notified = False
    for bot in targets:
        if bot.id not in thread_bot_ids:
            session.add(ThreadParticipant(thread_id=thread_id, kind="bot", bot_id=bot.id))
            thread_bot_ids.append(bot.id)
    extra_messages: list[Message] = []
    runs: list[Run] = []
    if targets and hop >= services.settings.max_bot_hops:
        if not thread.hop_limit_notified:
            notice = Message(thread_id=thread_id, sender_kind="system", sender_name="system",
                             content=HOP_LIMIT_NOTICE, created_at=utcnow(), meta={"kind": "hop_limit"})
            session.add(notice)
            extra_messages.append(notice)
            thread.hop_limit_notified = True
        targets = []
    for bot in targets:
        run = Run(bot_id=bot.id, thread_id=thread_id, trigger_message_id=msg.id, status="queued")
        session.add(run)
        runs.append(run)
    await session.commit()
    for m in [msg, *extra_messages]:
        if services.store is not None:
            await memory.index_message(services.store, m)
        await services.bus.publish("message.created", thread_id, to_json(MessageOut, m))
    for run in runs:
        await services.bus.publish("run.updated", thread_id, to_json(RunOut, run))
    if services.queue is not None:
        for run in runs:
            await services.queue.enqueue(run.id)
    return PostResult(message=msg, runs=runs, unaddressed=unaddressed)
```

```python
async def participants_for(session: AsyncSession, thread_ids: list[str]) -> dict[str, list[ParticipantOut]]:
    if not thread_ids:

        .where(ThreadParticipant.thread_id.in_(thread_ids)))).all()

                                               handle=bot.handle if bot else None, name=bot.name if bot else None))
    return out

    t = ThreadOut.model_validate(thread)
    t.participants = parts
    return t


@router.get("", response_model=list[ThreadOut])

    threads = (await session.execute(select(Thread).order_by(Thread.updated_at.desc()))).scalars().all()
    parts = await participants_for(session, [t.id for t in threads])
    return [thread_out(t, parts[t.id]) for t in threads]

                 services: Services = Depends(get_services)):

                                     created_by_kind="human")
    except ValueError as e:
        raise HTTPException(422, str(e))
    parts = await participants_for(session, [thread.id])
    return thread_out(thread, parts[thread.id])


@router.get("/{thread_id}", response_model=ThreadDetail)
async def get_thread(thread_id: str, before: str | None = None, limit: int = 50,
                     session: AsyncSession = Depends(get_session)):
    thread = await session.get(Thread, thread_id)
    if not thread:
        raise HTTPException(404, "thread not found")
    q = select(Message).where(Message.thread_id == thread_id)
    if before:
        anchor = await session.get(Message, before)
        if anchor:
            q = q.where(Message.created_at < anchor.created_at)
    rows = (await session.execute(q.order_by(Message.created_at.desc()).limit(limit + 1))).scalars().all()
    has_more = len(rows) > limit
    msgs = list(reversed(rows[:limit]))
    runs = (await session.execute(select(Run).where(Run.thread_id == thread_id, Run.status.in_(OPEN_RUN_STATUSES))
                                  .order_by(Run.created_at))).scalars().all()
    parts = await participants_for(session, [thread_id])
    detail = ThreadDetail.model_validate(thread)
    detail.participants = parts[thread_id]
    detail.messages = [MessageOut.model_validate(m) for m in msgs]
    detail.has_more = has_more
    detail.runs = [RunOut.model_validate(r) for r in runs]
    return detail


@router.delete("/{thread_id}", status_code=204)
async def delete_thread(thread_id: str, session: AsyncSession = Depends(get_session)):
    thread = await session.get(Thread, thread_id)
    if not thread:
        raise HTTPException(404, "thread not found")
    for model in (Run, Message, ThreadParticipant):
        for row in (await session.execute(select(model).where(model.thread_id == thread_id))).scalars():
            await session.delete(row)
    await session.delete(thread)
    await session.commit()
```

```python
# backend/openbot/runtime/prompt.py
from __future__ import annotations

from langchain_core.messages import AIMessage, BaseMessage, HumanMessage

from openbot.db.models import Bot, Message


def estimate_tokens(text: str) -> int:
    return len(text) // 4 + 8


def build_history(messages: list[Message], bot_id: str, *, token_budget: int,
                  max_messages: int) -> tuple[list[BaseMessage], int]:
    ordered = sorted(messages, key=lambda m: (m.created_at, m.id))
    picked: list[Message] = []
    used = 0
    for m in reversed(ordered):
        cost = estimate_tokens(m.content)
        if picked and (len(picked) >= max_messages or used + cost > token_budget):
            break
        picked.append(m)
        used += cost
    picked.reverse()
    older = len(ordered) - len(picked)
    out: list[BaseMessage] = []
    for m in picked:
        if m.sender_kind == "bot" and m.sender_bot_id == bot_id:
            out.append(AIMessage(content=m.content))
            continue
        line = f"[{m.sender_name}]: {m.content}"
        if out and isinstance(out[-1], HumanMessage):
            out[-1] = HumanMessage(content=f"{out[-1].content}\n\n{line}")
        else:
            out.append(HumanMessage(content=line))
    return out, older


def build_system_prompt(*, bot: Bot, all_bots: list[Bot], participants: list[str], memories: list[str],
                        workspace_root: str, older_count: int, tool_names: list[str]) -> str:
    roster = "\n".join(f"- @{b.handle} ({b.name}): {b.description or 'no description'}"
                       for b in all_bots if b.enabled and b.id != bot.id) or "- (no other bots)"
    mem = "\n".join(f"- {m}" for m in memories) or "- (none yet)"
    older = (f"This thread has {older_count} older messages not shown. Use recall_messages (semantic search) "
             f"or read_history (chronological paging) to fetch them.") if older_count else \
            "The full thread history is shown."
    return f"""You are {bot.name} (@{bot.handle}), a persistent AI bot on the OpenBot platform.
{bot.description}

{bot.instructions}

# How this platform works
- You are in a shared thread with: {', '.join(participants) or 'nobody else'}. Messages from others appear as "[name]: text".
- Your reply is posted to the thread as a message from you. To hand work to another bot or ask it something, mention it with @handle in your reply. Only mentioned bots are woken up. Never mention yourself.
```

```python
    services = ctx.services
    async with services.session_factory() as s:
        try:
            t = await create_thread(services, s, title=title, handles=handles, created_by_kind="bot",
                                    created_by_bot_id=ctx.bot_id)

@tool
def ask_human(question: str) -> str:
    """Ask the human a question or request a decision. Your run pauses until they answer."""
    answer = interrupt({"kind": "question", "question": question})
    return f"Human answered: {answer}"


def _fmt(m: Message) -> str:
    return f"{m.created_at.isoformat(timespec='seconds')} [{m.sender_name}]: {m.content}"


@tool
async def read_history(runtime: ToolRuntime[RunContext], before_message_id: str | None = None,
                       limit: int = 20) -> str:
    """Read older messages of the current thread, chronologically. Pass before_message_id to page further back."""
    ctx = runtime.context
    async with ctx.services.session_factory() as s:
        q = select(Message).where(Message.thread_id == ctx.thread_id)
        if before_message_id:
            anchor = await s.get(Message, before_message_id)
            if anchor:
                q = q.where(Message.created_at < anchor.created_at)
        rows = (await s.execute(q.order_by(Message.created_at.desc()).limit(limit))).scalars().all()
    rows.reverse()

                     for h in hits) or "(no matches)"
```

```python
# backend/openbot/runtime/runner.py
from __future__ import annotations

import asyncio
import logging
from typing import Any

from langchain.agents import create_agent
from langchain.agents.middleware import HumanInTheLoopMiddleware
from langchain_core.messages import AIMessage, AIMessageChunk, ToolMessage
from langchain_core.tracers.context import collect_runs
from langgraph.types import Command
from sqlalchemy import func, select

from openbot.api.schemas import RunEventOut, RunOut, to_json
from openbot.db.models import Bot, Message, Run, RunEvent, Thread, ThreadParticipant, utcnow
from openbot.runtime import memory
from openbot.runtime.messaging import post_message
from openbot.runtime.prompt import build_history, build_system_prompt
from openbot.tools.builtin.core import CORE_TOOLS
from openbot.tools.context import RunContext

log = logging.getLogger(__name__)
TOOL_RESULT_CAP = 4000


def normalize_interrupt(value: Any) -> dict:
    if isinstance(value, dict):
        if value.get("kind") == "question":
            return {"kind": "question", "question": str(value.get("question", ""))}
        if "action_requests" in value:
            return {"kind": "approval",
                    "actions": [{"name": a.get("name"), "args": a.get("args", {})} for a in value["action_requests"]]}
    return {"kind": "question", "question": str(value)}


class Runner:
    def __init__(self, services) -> None:
        self.s = services

    # ---- persistence helpers -------------------------------------------------
    async def _set_status(self, run_id: str, status: str, *, interrupt: dict | None = None,
                          error: str | None = None, langsmith_run_id: str | None = None) -> Run:
        async with self.s.session_factory() as session:
            run = await session.get(Run, run_id)
            run.status = status
            run.interrupt = interrupt
            if error is not None:
                run.error = error
            if langsmith_run_id:
                run.langsmith_run_id = langsmith_run_id
            if status == "running" and run.started_at is None:
                run.started_at = utcnow()
            if status in ("completed", "failed", "cancelled"):
                run.finished_at = utcnow()
            await session.commit()
            await self.s.bus.publish("run.updated", run.thread_id, to_json(RunOut, run))
            return run

    async def _record(self, run: Run, seq: int, type_: str, payload: dict) -> int:
        async with self.s.session_factory() as session:
            ev = RunEvent(run_id=run.id, seq=seq, type=type_, payload=payload)
            session.add(ev)
            await session.commit()
            await self.s.bus.publish("run.event", run.thread_id, to_json(RunEventOut, ev))
        return seq + 1

    async def _next_seq(self, run_id: str) -> int:
        async with self.s.session_factory() as session:
            n = (await session.execute(select(func.max(RunEvent.seq)).where(RunEvent.run_id == run_id))).scalar()
        return 0 if n is None else n + 1

    async def _system_message(self, thread_id: str, content: str) -> None:
        async with self.s.session_factory() as session:
            await post_message(self.s, session, thread_id=thread_id, content=content, sender_kind="system",
                               sender_name="system")

    # ---- agent construction ---------------------------------------------------
    async def _prepare(self, bot: Bot, thread: Thread, trigger: Message | None) -> tuple[str, dict, list[str]]:
        st = self.s.settings
        async with self.s.session_factory() as session:
            all_bots = (await session.execute(select(Bot))).scalars().all()
            rows = (await session.execute(
                select(Message).where(Message.thread_id == thread.id).order_by(Message.created_at.desc())
                .limit(st.history_max_messages * 2))).scalars().all()
            total = (await session.execute(select(func.count()).select_from(Message)
                                           .where(Message.thread_id == thread.id))).scalar() or 0
            parts = (await session.execute(select(ThreadParticipant).where(ThreadParticipant.thread_id == thread.id))).scalars().all()
        history, older = build_history(list(rows), bot.id, token_budget=st.history_token_budget,
                                       max_messages=st.history_max_messages)
        older += max(0, total - len(rows))
        by_id = {b.id: b for b in all_bots}
        participants = [by_id[p.bot_id].name if p.bot_id in by_id else p.kind for p in parts]
        query = trigger.content if trigger else ""
        memories = await memory.relevant_memories(self.s.store, bot.id, query) if self.s.store else []
        system_prompt = build_system_prompt(bot=bot, all_bots=list(all_bots), participants=participants,
                                            memories=memories, workspace_root=str(st.workspace_root),
                                            older_count=older, tool_names=list(bot.tool_names))
        return system_prompt, {"messages": history}, [b.handle for b in all_bots]

    def _build_agent(self, bot: Bot, system_prompt: str):
        tools = [*self.s.registry.resolve(list(bot.tool_names)), *CORE_TOOLS,
                 *memory.memory_tools(bot.id, self.s.store)]
        middleware = []
        if bot.approval_tools:
            middleware.append(HumanInTheLoopMiddleware(
                interrupt_on={t: {"allowed_decisions": ["approve", "reject"]} for t in bot.approval_tools},
                description_prefix="Tool execution requires approval"))
        model = self.s.model_factory(bot)
        return create_agent(model, tools=tools, system_prompt=system_prompt, middleware=middleware,
                            checkpointer=self.s.checkpointer, store=self.s.store, context_schema=RunContext)

    # ---- streaming ------------------------------------------------------------
    async def _stream(self, agent, inputs, config, ctx: RunContext, run: Run, seq: int) -> tuple[str, dict | None, int]:
        final_text = ""
        interrupt: dict | None = None
        async for mode, data in agent.astream(inputs, config=config, context=ctx, stream_mode=["messages", "updates"]):
            if mode == "messages":
                token, meta = data
                if isinstance(token, AIMessageChunk) and token.text and meta.get("langgraph_node") == "model":
                    await self.s.bus.publish("run.event", run.thread_id,
                                             {"run_id": run.id, "type": "text_delta", "payload": {"delta": token.text}})
                continue
            for source, update in data.items():
                if source == "__interrupt__":
                    interrupt = normalize_interrupt(update[0].value)
                    seq = await self._record(run, seq, "interrupt", interrupt)
                elif source == "model" and isinstance(update, dict):
                    for m in update.get("messages", []):
                        if not isinstance(m, AIMessage):
                            continue
                        for tc in m.tool_calls:
                            seq = await self._record(run, seq, "tool_call",
                                                     {"id": tc["id"], "name": tc["name"], "args": tc["args"]})
                        if m.text:
                            final_text = m.text
                            seq = await self._record(run, seq, "text", {"content": m.text})
                elif source == "tools" and isinstance(update, dict):
                    for m in update.get("messages", []):
                        if isinstance(m, ToolMessage):
                            content = m.text if hasattr(m, "text") else str(m.content)
                            seq = await self._record(run, seq, "tool_result", {
                                "tool_call_id": m.tool_call_id, "name": m.name, "status": m.status,
                                "content": content[:TOOL_RESULT_CAP]})
        return final_text, interrupt, seq

    # ---- main entry -------------------------------------------------------------
    async def execute(self, run_id: str, resume: Command | None = None) -> None:
        async with self.s.session_factory() as session:
            run = await session.get(Run, run_id)
            if run is None or run.status not in ("queued", "waiting_human"):
                return
            bot = await session.get(Bot, run.bot_id)
            thread = await session.get(Thread, run.thread_id)
            trigger = await session.get(Message, run.trigger_message_id) if run.trigger_message_id else None
        await self._set_status(run.id, "running")
        seq = await self._next_seq(run.id)
        if resume is not None:
            seq = await self._record(run, seq, "resumed", {"value": getattr(resume, "resume", None)})
        hop = (trigger.hop + 1) if trigger else 1
        ctx = RunContext(bot.id, bot.handle, bot.name, thread.id, run.id, self.s.settings.workspace_root,
                         self.s, hop)
        config = {"configurable": {"thread_id": run.id},
                  "metadata": {"bot": bot.handle, "thread_id": thread.id, "run_id": run.id},
                  "run_name": f"bot:{bot.handle}"}
        agent = None
        try:
            system_prompt, inputs, _ = await self._prepare(bot, thread, trigger)
            agent = self._build_agent(bot, system_prompt)
            with collect_runs() as cb:
                final_text, interrupt, seq = await self._stream(agent, resume if resume is not None else inputs,
                                                                config, ctx, run, seq)
            ls_id = str(cb.traced_runs[0].id) if cb.traced_runs else None
            if interrupt is not None:
                await self._set_status(run.id, "waiting_human", interrupt=interrupt, langsmith_run_id=ls_id)
                return
            if final_text.strip():
                async with self.s.session_factory() as session:
                    res = await post_message(self.s, session, thread_id=thread.id, content=final_text,
                                             sender_kind="bot", sender_bot_id=bot.id, sender_name=bot.name,
                                             hop=hop, run_id=run.id)
                seq = await self._record(run, seq, "message", {"message_id": res.message.id})
            await self._set_status(run.id, "completed", langsmith_run_id=ls_id)
            if bot.memory_enabled and self.s.reflector is not None:
                state = await agent.aget_state(config)
                self.s.reflector.schedule(bot, list(state.values.get("messages", [])))
        except asyncio.CancelledError:
            await self._set_status(run.id, "cancelled")
            await self._system_message(thread.id, f"@{bot.handle} run was cancelled.")
            raise
        except Exception as e:  # noqa: BLE001
            log.exception("run %s failed", run.id)
            err = f"{type(e).__name__}: {e}"[:2000]
            await self._record(run, seq, "error", {"error": err})
            await self._set_status(run.id, "failed", error=err)
            await self._system_message(thread.id, f"@{bot.handle} failed: {err}")
        finally:
            if self.s.queue is not None:
                await self.s.queue.kick(bot.id, thread.id)
```

```python
from __future__ import annotations

import asyncio
import logging
from typing import Protocol

from langgraph.types import Command
from sqlalchemy import select

from openbot.api.schemas import RunOut, to_json
from openbot.db.models import ACTIVE_RUN_STATUSES, Run, utcnow

log = logging.getLogger(__name__)


class RunQueue(Protocol):
    async def enqueue(self, run_id: str, resume: Command | None = None) -> None: ...
    async def kick(self, bot_id: str, thread_id: str) -> None: ...
    async def cancel(self, run_id: str) -> bool: ...
    async def shutdown(self) -> None: ...


class InProcessRunQueue:
    def __init__(self, services, max_concurrent: int) -> None:
        self.s = services
        self._sem = asyncio.Semaphore(max_concurrent)
        self._tasks: dict[str, asyncio.Task] = {}
        self._pair_lock = asyncio.Lock()

    async def enqueue(self, run_id: str, resume: Command | None = None) -> None:
        if run_id in self._tasks:
            return
        task = asyncio.create_task(self._run(run_id, resume), name=f"run:{run_id}")
        self._tasks[run_id] = task
        task.add_done_callback(lambda t: self._tasks.pop(run_id, None))

    async def _claim(self, run_id: str) -> bool:
        async with self._pair_lock:
            async with self.s.session_factory() as session:
                run = await session.get(Run, run_id)
                if run is None or run.status != "queued":
                    return False
                active = (await session.execute(select(Run.id).where(
                    Run.bot_id == run.bot_id, Run.thread_id == run.thread_id,
                    Run.status.in_(ACTIVE_RUN_STATUSES), Run.id != run.id))).first()
                if active:
                    return False
                older = (await session.execute(select(Run).where(
                    Run.bot_id == run.bot_id, Run.thread_id == run.thread_id, Run.status == "queued",
                    Run.created_at < run.created_at))).scalars().all()
                superseded = []
                for o in older:
                    o.status, o.error, o.finished_at = "cancelled", "superseded", utcnow()
                    superseded.append(o)
                await session.commit()
                for o in superseded:
                    await self.s.bus.publish("run.updated", o.thread_id, to_json(RunOut, o))
                return True

    async def _run(self, run_id: str, resume: Command | None) -> None:
        async with self._sem:
            if resume is None and not await self._claim(run_id):
                return
            try:
                await self.s.runner.execute(run_id, resume=resume)
            except asyncio.CancelledError:
                pass
            except Exception:  # noqa: BLE001
                log.exception("run task %s crashed", run_id)

    async def kick(self, bot_id: str, thread_id: str) -> None:
        async with self.s.session_factory() as session:
            active = (await session.execute(select(Run.id).where(
                Run.bot_id == bot_id, Run.thread_id == thread_id, Run.status.in_(ACTIVE_RUN_STATUSES)))).first()
            if active:
                return
            newest = (await session.execute(select(Run).where(
                Run.bot_id == bot_id, Run.thread_id == thread_id, Run.status == "queued")
                .order_by(Run.created_at.desc()).limit(1))).scalar_one_or_none()
        if newest is not None and newest.id not in self._tasks:
            await self.enqueue(newest.id)

    async def cancel(self, run_id: str) -> bool:
        task = self._tasks.get(run_id)
        if task and not task.done():
            task.cancel()
            return True
        async with self.s.session_factory() as session:
            run = await session.get(Run, run_id)
            if run is None or run.status not in ("queued", "waiting_human"):
                return False
            run.status, run.finished_at = "cancelled", utcnow()
            await session.commit()
            await self.s.bus.publish("run.updated", run.thread_id, to_json(RunOut, run))
        return True

    async def wait_idle(self) -> None:
        while self._tasks:
            await asyncio.gather(*list(self._tasks.values()), return_exceptions=True)

    async def shutdown(self) -> None:
        for t in list(self._tasks.values()):
            t.cancel()
        await asyncio.gather(*self._tasks.values(), return_exceptions=True)
```

```python
async def get_run(run_id: str, session: AsyncSession = Depends(get_session)):
    run = await session.get(Run, run_id)
    if not run:
        raise HTTPException(404, "run not found")
    events = (await session.execute(select(RunEvent).where(RunEvent.run_id == run_id).order_by(RunEvent.seq))).scalars().all()
    d = RunDetail.model_validate(run)
    d.events = [RunEventOut.model_validate(e) for e in events]
    return d


@router.post("/{run_id}/resume", response_model=RunOut)
async def resume_run(run_id: str, body: ResumeBody, session: AsyncSession = Depends(get_session),
                     services: Services = Depends(get_services)):
    run = await session.get(Run, run_id)
    if not run:
        raise HTTPException(404, "run not found")
    if run.status != "waiting_human" or not run.interrupt:
        raise HTTPException(409, "run is not waiting for human input")
    kind = run.interrupt.get("kind")
    if kind == "question":
        if body.answer is None:
            raise HTTPException(422, "answer is required for a question")
        cmd = Command(resume=body.answer)
    else:
        n = len(run.interrupt.get("actions", []))
        if not body.decisions or len(body.decisions) != n:
            raise HTTPException(422, f"decisions must have exactly {n} entries")
        cmd = Command(resume={"decisions": [{"type": d} for d in body.decisions]})
    await services.queue.enqueue(run.id, resume=cmd)
    return run


@router.post("/{run_id}/cancel", response_model=RunOut)
async def cancel_run(run_id: str, session: AsyncSession = Depends(get_session),
                     services: Services = Depends(get_services)):
    run = await session.get(Run, run_id)
    if not run:
        raise HTTPException(404, "run not found")
    if not await services.queue.cancel(run_id):
        raise HTTPException(409, "run is not cancellable")
    await session.refresh(run)
    return run
```

```python
    for r in services._owned_resources or []:
        if isinstance(r, AsyncExitStack):
            await r.aclose()

            await r.dispose()


async def start_background(services: Services) -> None:
    async with services.session_factory() as session:
        stale = (await session.execute(select(Run).where(Run.status == "running"))).scalars().all()
        for r in stale:
            r.status, r.error, r.finished_at = "failed", "server restarted", utcnow()
        queued = (await session.execute(select(Run.bot_id, Run.thread_id).where(Run.status == "queued").distinct())).all()
        await session.commit()
    if services.dispatcher is not None:
        await services.dispatcher.start()
    if services.settings.seed_demo_bots:
        await seed_demo_bots(services)
    for bot_id, thread_id in queued:
        await services.queue.kick(bot_id, thread_id)

    if services.queue is not None:
        await services.queue.shutdown()
    if services.reflector is not None:
        await services.reflector.shutdown()
    if services.dispatcher is not None:
        await services.dispatcher.stop()


def create_app(settings: Settings | None = None, services: Services | None = None) -> FastAPI:
    load_dotenv()
    settings = settings or get_settings()

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        owned = app.state.services is None
        if owned:
            app.state.services = await build_services(settings)
        await start_background(app.state.services)
        try:
            yield
        finally:
            await stop_background(app.state.services)
            if owned:
                await close_services(app.state.services)

    app = FastAPI(title="OpenBot", lifespan=lifespan)
    app.state.settings = settings
    app.state.services = services
    app.add_middleware(CORSMiddleware, allow_origins=settings.cors_origins, allow_methods=["*"], allow_headers=["*"])

              webhooks.router, events.router):

    app.include_router(public)
    app.include_router(api)

    dist = settings.frontend_dist
    if dist and Path(dist).is_dir():
        app.mount("/", StaticFiles(directory=str(dist), html=True), name="frontend")
    return app
```

```python
# backend/openbot/runtime/webhooks.py
from __future__ import annotations

import asyncio
import hashlib
import hmac
import json
import logging

import httpx
from sqlalchemy import select

from openbot.db.models import Webhook, WebhookDelivery, utcnow

log = logging.getLogger(__name__)


def sign(secret: str, body: bytes) -> str:
    return "sha256=" + hmac.new(secret.encode(), body, hashlib.sha256).hexdigest()


def map_event(item: dict) -> str | None:
    event, data = item["event"], item["data"]
    if event == "message.created" and data.get("sender_kind") == "bot":
        return "message.created"
    if event == "run.updated" and data.get("status") in ("waiting_human", "completed", "failed"):
        return f"run.{data['status']}"
    return None


class WebhookDispatcher:
    def __init__(self, services, http_client: httpx.AsyncClient | None = None) -> None:
        self.s = services
        self.http = http_client or httpx.AsyncClient(timeout=15)
        self._task: asyncio.Task | None = None
        self._inflight: set[asyncio.Task] = set()

    async def start(self) -> None:
        self._task = asyncio.create_task(self._loop(), name="webhook-dispatcher")

    async def stop(self) -> None:
        if self._task:
            self._task.cancel()
            await asyncio.gather(self._task, return_exceptions=True)
        for t in self._inflight:
            t.cancel()
        await asyncio.gather(*self._inflight, return_exceptions=True)
        await self.http.aclose()

    async def wait_idle(self) -> None:
        await asyncio.sleep(0)
        while self._inflight:
            await asyncio.gather(*list(self._inflight), return_exceptions=True)

    async def _loop(self) -> None:
        async with self.s.bus.subscribe(None) as q:
            while True:
                item = await q.get()
                name = map_event(item)
                if name is None:
                    continue
                async with self.s.session_factory() as session:
                    hooks = (await session.execute(select(Webhook).where(Webhook.enabled == True))).scalars().all()  # noqa: E712
                for hook in hooks:
                    if not self._matches(hook, name, item):
                        continue
                    t = asyncio.create_task(self._deliver(hook, name, item))
                    self._inflight.add(t)
                    t.add_done_callback(self._inflight.discard)

    @staticmethod
    def _matches(hook: Webhook, name: str, item: dict) -> bool:
        if name not in hook.events:
            return False
        if hook.thread_id and hook.thread_id != item.get("thread_id"):
            return False
        data = item["data"]
        if hook.bot_id and hook.bot_id not in (data.get("bot_id"), data.get("sender_bot_id")):
            return False
        return True

    async def _deliver(self, hook: Webhook, name: str, item: dict) -> None:
        async with self.s.session_factory() as session:
            d = WebhookDelivery(webhook_id=hook.id, event=name, payload=item["data"])
            session.add(d)
            await session.commit()
            delivery_id = d.id
        body = json.dumps({"event": name, "delivery_id": delivery_id, "thread_id": item.get("thread_id"),
                           "data": item["data"], "created_at": utcnow().isoformat()}, default=str).encode()
        headers = {"Content-Type": "application/json", "X-OpenBot-Event": name,
                   "X-OpenBot-Delivery": delivery_id, "X-OpenBot-Signature": sign(hook.secret, body)}
        delays = [0.0, *self.s.settings.webhook_retry_delays]
        last_error = None
        for attempt, delay in enumerate(delays, start=1):
            if delay:
                await asyncio.sleep(delay)
            try:
                r = await self.http.post(hook.url, content=body, headers=headers)
                if r.status_code < 300:
                    await self._finish(delivery_id, "delivered", attempt, None)
                    return
                last_error = f"HTTP {r.status_code}"
            except httpx.HTTPError as e:
                last_error = f"{type(e).__name__}: {e}"
            await self._finish(delivery_id, "pending", attempt, last_error)
        await self._finish(delivery_id, "failed", len(delays), last_error)

    async def _finish(self, delivery_id: str, status: str, attempts: int, error: str | None) -> None:
        async with self.s.session_factory() as session:
            d = await session.get(WebhookDelivery, delivery_id)
            d.status, d.attempts, d.last_error = status, attempts, error
            if status == "delivered":
                d.delivered_at = utcnow()
            await session.commit()
```

```python
from sqlalchemy.ext.asyncio import AsyncSession

router = APIRouter(prefix="/webhooks", tags=["webhooks"])
VALID_EVENTS = {"message.created", "run.waiting_human", "run.completed", "run.failed"}


@router.get("", response_model=list[WebhookOut])
async def list_webhooks(session: AsyncSession = Depends(get_session)):
    return (await session.execute(select(Webhook).order_by(Webhook.created_at))).scalars().all()


@router.post("", response_model=WebhookOut, status_code=201)
async def create_webhook(body: WebhookCreate, session: AsyncSession = Depends(get_session)):
    bad = set(body.events) - VALID_EVENTS
    if bad:
        raise HTTPException(422, f"unknown events: {sorted(bad)}")
    hook = Webhook(**body.model_dump())
    session.add(hook)
    await session.commit()
    return hook

    await session.delete(hook)
    await session.commit()
```

```python
async def seed_demo_bots(services) -> int:
    dp = default_provider(services.settings)
    if dp is None:
        log.info("no provider configured; skipping demo bot seed")
        return 0
    provider, model = dp
    async with services.session_factory() as session:
        if (await session.execute(select(Bot.id).limit(1))).first():
            return 0
        for spec in DEMO_BOTS:
            session.add(Bot(provider=provider, model=model, **spec))
        await session.commit()

    return len(DEMO_BOTS)
```
